def f(x, y):
    return x + (2 * (y ** 2))


print('Enter initial condition ')
x0 = float(input('x0 = '))
y0 = float(input('y0 = '))

print('Enter Calculation Points: ')
xn = float(input('xn = '))

print('Enter Number of Steps = ')
step = int(input('Number of Steps = '))


def rk4(x0, y0, xn, n):
    h = (xn - x0) / n

    print('\nSOLUTION')
    print('x\t\ty\t\ty_next')

    for i in range(n):
        k1 = h * f(x0, y0)
        k2 = h * f(x0 + h/2, y0 + k1/2)
        k3 = h * f(x0 + h/2, y0 + k2/2)
        k4 = h * f(x0 + h, y0 + k3)

        yn = y0 + (1/6) * (k1 + 2*k2 + 2*k3 + k4)

        print('%.4f\t%.4f\t%.4f' % (x0, y0, yn))

        y0 = yn
        x0 = x0 + h

    print('\nFinal result: At x = %.4f, y = %.4f' % (xn, y0))


rk4(x0, y0, xn, step)
